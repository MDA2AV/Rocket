The shared-nothing io_uring
runtime for .NET

One ring per reactor, one reactor per thread - run one per core. Every await - an HTTP read, a Postgres query, a file read - is a completion on that ring, resumed inline on the same thread. No thread pool on the hot path. The ring is the I/O.

A normal async handler. An abnormal runtime under it.

Ordinary C#

Every await is a request placed on this core's queue, and your code resumes where it left off when the kernel answers.

Nothing scheduled

Each await parks on a reusable IValueTaskSource. The completion arrives, the reactor calls SetResult, your code runs — same thread, no queue.

Both directions

A Postgres query, a Redis command, an outbound HTTP call or a file read rides the same ring that accepted the request.

All of it runs: Playground is a complete server per file, Examples has benchmarks, and Examples.AspNet measures UseIoxide() against stock Kestrel.

Pick an example to see the code

// A complete ioxide server. Paste into a console app's Program.cs and run it.
//   dotnet add package ioxide
using System.Buffers;
using System.IO.Pipelines;
using ioxide;
using ioxide.utils;

var config = new ServerConfig
{
    ReactorCount = Environment.ProcessorCount,
    Tcp = new TcpOptions { Port = 8080 },
};

ReadOnlyMemory<byte> ok = "HTTP/1.1 200 OK\r\nContent-Type: text/plain\r\nContent-Length: 2\r\n\r\nok"u8.ToArray();

var threads = new Thread[config.ReactorCount];

for (int id = 0; id < threads.Length; id++)
{
    var reactor = new Reactor(id, config);

    reactor.TcpHandle = async (r, conn) =>
    {
        // Zero-copy pipes over the ring. The reader owns the carry: unconsumed bytes are held
        // across reads (no copy, no slab) and buffers return to the ring automatically once
        // fully consumed. The writer stages into the connection's write slab.
        var reader = new TcpConnectionPipeReader(conn);
        var writer = new TcpConnectionPipeWriter(conn);

        try
        {
            while (true)
            {
                // io_uring recv behind the PipeReader surface - still resumes inline.
                ReadResult result = await reader.ReadAsync();
                ReadOnlySequence<byte> buffer = result.Buffer;   // every unconsumed byte so far

                // Walk every complete request; a partial one stays buffered for the next read.
                // Framing is your code - ioxide hands you bytes and stays out of HTTP.
                SequencePosition consumed = buffer.Start;
                bool respond = false;

                while (TryParseRequest(ref buffer, out _))
                {
                    consumed = buffer.Start;
                    respond = true;
                }

                if (respond)
                {
                    ok.Span.CopyTo(writer.GetSpan(ok.Length));
                    writer.Advance(ok.Length);
                    await writer.FlushAsync();
                }

                // Consumed bytes release their buffers; the partial tail is kept.
                reader.AdvanceTo(consumed, buffer.End);

                if (result.IsCompleted) return;
            }
        }
        finally
        {
            reader.Complete();
            conn.DecRef();
        }
    };

    threads[id] = new Thread(reactor.Run) { Name = $"reactor-{id}" };
    threads[id].Start();
}

Console.WriteLine($"listening on :{config.Tcp.Port} across {config.ReactorCount} reactors");

foreach (Thread thread in threads)
{
    thread.Join();
}

// Slice one complete request (terminated by a blank line) off the front of the buffer.
static bool TryParseRequest(ref ReadOnlySequence<byte> buffer, out ReadOnlySequence<byte> request)
{
    var reader = new SequenceReader<byte>(buffer);

    if (!reader.TryReadTo(out request, "\r\n\r\n"u8))
    {
        return false;
    }

    buffer = buffer.Slice(reader.Position);
    return true;
}

// A complete ioxide server. Paste into a console app's Program.cs and run it.
//   dotnet add package ioxide
using System.Buffers;
using ioxide;
using ioxide.utils;

var config = new ServerConfig
{
    ReactorCount = Environment.ProcessorCount,
    RingEntries  = 8192,                   // io_uring submission/completion queue depth

    // The shared recv ring: one pool of fixed buffers per reactor, handed out by the kernel.
    RecvBufferSize = 32 * 1024,            // size of each receive buffer
    RecvSlots      = 4096,                 // buffers in the shared recv ring

    Tcp = new TcpOptions
    {
        Port             = 8080,
        WriteSlabSize    = 16 * 1024,      // per-connection write buffer
        PoolMax          = 1024,           // connection objects recycled per reactor
        RecvQueueEntries = 64,             // per-connection queue of received slices
    },
};

ReadOnlyMemory<byte> ok = "HTTP/1.1 200 OK\r\nContent-Type: text/plain\r\nContent-Length: 2\r\n\r\nok"u8.ToArray();

var threads = new Thread[config.ReactorCount];

for (int id = 0; id < threads.Length; id++)
{
    // Each reactor owns its ring, its SO_REUSEPORT listener and its connections outright.
    var reactor = new Reactor(id, config);

    reactor.TcpHandle = async (r, conn) =>
    {
        try
        {
            while (true)
            {
                // io_uring recv. Resumes inline on this reactor thread - no thread pool hop.
                RecvSnapshot snapshot = await conn.ReadAsync();

                // A snapshot can carry several buffers. Each one is a slice of the shared ring:
                // read what you need, then hand it straight back so the kernel can reuse it.
                while (conn.TryGetItem(snapshot, out SpscRecvRing.Item item))
                {
                    if (item.HasBuffer)
                    {
                        ReadOnlySpan<byte> received = item.AsSpan();   // your bytes, still in the ring
                        _ = received.Length;                           // parse here - ioxide stays out of HTTP

                        conn.ReturnBuffer(in item);
                    }
                }

                conn.Write(ok.Span);
                await conn.FlushAsync();

                if (snapshot.IsClosed) return;   // peer went away
                conn.ResetRead();                // arm the next read
            }
        }
        finally
        {
            conn.DecRef();   // hand the connection object back to the reactor's pool
        }
    };

    threads[id] = new Thread(reactor.Run) { Name = $"reactor-{id}" };
    threads[id].Start();
}

Console.WriteLine($"listening on :{config.Tcp.Port} across {config.ReactorCount} reactors");

foreach (Thread thread in threads)
{
    thread.Join();
}

TCP · raw, incremental ring

// A complete ioxide server. Paste into a console app's Program.cs and run it.
//   dotnet add package ioxide
//   Needs Linux 6.12+ for incremental buffer rings.
using ioxide;
using ioxide.utils;

var config = new ServerConfig
{
    ReactorCount = Environment.ProcessorCount,
    RingEntries  = 8192,

    // Incremental (kernel 6.12+): a buffer ring PER CONNECTION, which the kernel appends into
    // across successive recvs. A large request lands contiguously instead of arriving as a chain
    // of shared-ring slices you have to stitch together. The shared-ring knobs go unused.
    Incremental = new IncrementalOptions
    {
        MaxConnections = 4096,   // one buffer-group id each; accepts past this are shed
        RecvSlots      = 16,     // buffers per connection ring
        RecvBufferSize = 4096,   // bytes per buffer - the kernel appends into it across recvs
    },

    Tcp = new TcpOptions
    {
        Port             = 8080,
        WriteSlabSize    = 16 * 1024,
        PoolMax          = 1024,
        RecvQueueEntries = 64,
    },
};

ReadOnlyMemory<byte> ok = "HTTP/1.1 200 OK\r\nContent-Type: text/plain\r\nContent-Length: 2\r\n\r\nok"u8.ToArray();

var threads = new Thread[config.ReactorCount];

for (int id = 0; id < threads.Length; id++)
{
    var reactor = new Reactor(id, config);

    reactor.TcpHandle = async (r, conn) =>
    {
        try
        {
            while (true)
            {
                RecvSnapshot snapshot = await conn.ReadAsync();

                // Same drain loop as the shared-ring flavour - the difference is that consecutive
                // recvs for THIS connection land in its own buffers, so a request that spans
                // several reads stays contiguous rather than scattering across the shared pool.
                while (conn.TryGetItem(snapshot, out SpscRecvRing.Item item))
                {
                    if (item.HasBuffer)
                    {
                        ReadOnlySpan<byte> received = item.AsSpan();
                        _ = received.Length;   // parse here

                        conn.ReturnBuffer(in item);
                    }
                }

                conn.Write(ok.Span);
                await conn.FlushAsync();

                if (snapshot.IsClosed) return;
                conn.ResetRead();
            }
        }
        finally
        {
            conn.DecRef();
        }
    };

    threads[id] = new Thread(reactor.Run) { Name = $"reactor-{id}" };
    threads[id].Start();
}

Console.WriteLine($"listening on :{config.Tcp.Port} across {config.ReactorCount} reactors (incremental)");

foreach (Thread thread in threads)
{
    thread.Join();
}

Shared ring vs incremental

Shared buffer ring (default, kernel 6.1+): one pool of buffers per reactor, drawn on by every connection. One recv consumes one whole buffer no matter how few bytes arrived - simple and elastic across connections, but small messages waste buffer space, and a buffer-hoarding connection eats from everyone's pool.

Incremental (kernel 6.12+): a small buffer ring per connection, and the kernel keeps appending successive recvs into the same buffer until it fills. Small messages pack densely and every connection's memory is isolated and bounded - at the cost of refcounted recycling (a buffer returns once you and the kernel are both done with it) and a ring registration per connection.

sharedincremental
buffer ownershipone pool per reactorone small ring per connection
fill behaviorone recv = one whole bufferrecvs append into the same buffer
best formedium/large messages, simplicitymany connections, small messages
memory shapeelastic, sharedisolated, bounded per connection
return pathpush the id backrefcounted: you + kernel both done
kernel6.1+6.12+

The handler code is identical in both modes - set Incremental to an IncrementalOptions (or null for the shared ring); ReturnBuffers routes the right return path internally. The pipes tab is API sugar over either mode: the reader owns the carry for you.

// A complete ioxide QUIC server. Paste into a console app's Program.cs and run it.
//   dotnet add package ioxide
//   dotnet add package ioxide.ngtcp2
// Needs cert.pem / key.pem next to the binary.
using System.Buffers;
using System.IO.Pipelines;
using ioxide;
using ioxide.ngtcp2;

// One engine for the whole server: ngtcp2 + picotls, compiled into the package. TLS 1.3 lives
// inside the transport, so there is no separate TLS layer to wire up.
using var engine = new QuicEngine("cert.pem", "key.pem", cidLength: 8,
                                  alpn: ["echo"]);   // ALPN allowlist - ["h3"] when serving HTTP/3

var config = new ServerConfig
{
    ReactorCount = Environment.ProcessorCount,
    Udp = new UdpOptions { RecvSlots = 16 },   // multishot UDP recv slots per reactor
    Quic = new QuicOptions
    {
        Port              = 8443,              // every reactor binds it via SO_REUSEPORT
        LocalCidLength    = 8,
        ConnectionFactory = engine.CreateFactory(),
    },
};

var threads = new Thread[config.ReactorCount];

for (int id = 0; id < threads.Length; id++)
{
    var reactor = new Reactor(id, config);

    reactor.QuicHandle = async (r, conn) =>
    {
        // One QUIC stream as a duplex pipe. QUIC multiplexes streams and a PipeReader is one byte
        // stream, so the pipe binds to a single stream: the first one the peer opens (auto-bind),
        // or an explicit id. Multi-stream protocols use the raw surface - that is what
        // ioxide.nghttp3 does.
        var pipe = new QuicConnectionDualPipe(conn);

        while (true)
        {
            // Decrypted stream bytes - resumed inline on the reactor.
            ReadResult result = await pipe.Input.ReadAsync();
            ReadOnlySequence<byte> buffer = result.Buffer;

            foreach (ReadOnlyMemory<byte> segment in buffer)
            {
                await pipe.Output.WriteAsync(segment);   // echo it straight back
            }

            pipe.Input.AdvanceTo(buffer.End);

            if (result.IsCompleted) return;
        }
    };

    threads[id] = new Thread(reactor.Run) { Name = $"reactor-{id}" };
    threads[id].Start();
}

Console.WriteLine($"QUIC on udp :8443 across {config.ReactorCount} reactors (ngtcp2 {QuicEngine.NativeVersion()})");

foreach (Thread thread in threads)
{
    thread.Join();
}

QUIC · raw streams

ioxide + ioxide.ngtcp2 
// A complete ioxide QUIC server. Paste into a console app's Program.cs and run it.
//   dotnet add package ioxide
//   dotnet add package ioxide.ngtcp2
// Needs cert.pem / key.pem next to the binary.
using ioxide;
using ioxide.ngtcp2;

using var engine = new QuicEngine("cert.pem", "key.pem", cidLength: 8, alpn: ["echo"]);

var config = new ServerConfig
{
    ReactorCount = Environment.ProcessorCount,
    Udp = new UdpOptions { RecvSlots = 16 },
    Quic = new QuicOptions
    {
        Port              = 8443,
        LocalCidLength    = 8,
        ConnectionFactory = engine.CreateFactory(),
    },
};

var threads = new Thread[config.ReactorCount];

for (int id = 0; id < threads.Length; id++)
{
    var reactor = new Reactor(id, config);

    reactor.QuicHandle = async (r, conn) =>
    {
        // The raw surface: every stream on the connection, not just one. A delivery carries the
        // stream id alongside the bytes, so a multi-stream protocol can demux here - which is
        // exactly what ioxide.nghttp3 does on top of this.
        while (true)
        {
            QuicRecvSnapshot snapshot = await conn.ReadAsync();

            while (conn.TryGetDelivery(in snapshot, out QuicRecvRing.Delivery delivery))
            {
                ReadOnlySpan<byte> data = delivery.AsSpan();   // decrypted stream bytes

                // Echo on the same stream. fin: false keeps it open for more.
                conn.SendStream(delivery.StreamId, data, fin: false);

                conn.ReturnBuffer(in delivery);   // back to the ring
            }

            if (snapshot.IsClosed) return;
            conn.ResetRead();
        }
    };

    threads[id] = new Thread(reactor.Run) { Name = $"reactor-{id}" };
    threads[id].Start();
}

Console.WriteLine($"QUIC on udp :8443 across {config.ReactorCount} reactors (ngtcp2 {QuicEngine.NativeVersion()})");

foreach (Thread thread in threads)
{
    thread.Join();
}

HTTP/3 · nghttp3

ioxide + ioxide.ngtcp2 + ioxide.nghttp3 
// A complete HTTP/3 server. Paste into a console app's Program.cs and run it.
//   dotnet add package ioxide
//   dotnet add package ioxide.ngtcp2
//   dotnet add package ioxide.nghttp3
// Needs cert.pem / key.pem next to the binary.
//   curl --http3-only -k https://127.0.0.1:8443/
using ioxide;
using ioxide.nghttp3;
using ioxide.ngtcp2;

// ngtcp2 + picotls, bundled native. ALPN pinned to h3, so nothing else negotiates.
using var engine = new QuicEngine("cert.pem", "key.pem", cidLength: 8, alpn: ["h3"]);

var config = new ServerConfig
{
    ReactorCount = Environment.ProcessorCount,
    Udp = new UdpOptions { RecvSlots = 16 },
    Quic = new QuicOptions
    {
        Port              = 8443,          // every reactor binds it via SO_REUSEPORT
        LocalCidLength    = 8,
        ConnectionFactory = engine.CreateFactory(),
    },
};

// Build the response ONCE and reuse the instance. The h3 layer copies status, headers and body
// into nghttp3 synchronously at submit and never retains the object, so this costs zero
// allocations per request - unlike Nghttp3Response.Text($"..."), which encodes a fresh string
// every time.
var response = new Nghttp3Response { Body = "Hello, World!"u8.ToArray() };
response.Headers.Add("content-type"u8.ToArray(), "text/plain"u8.ToArray());
response.Headers.Add("server"u8.ToArray(), "ioxide"u8.ToArray());

var threads = new Thread[config.ReactorCount];

for (int id = 0; id < threads.Length; id++)
{
    var reactor = new Reactor(id, config);

    // Nghttp3Connection owns the connection's read loop: it feeds stream data into nghttp3,
    // assembles each request - headers, body and all - and calls your function once per request.
    // QPACK, control streams, fin and stream teardown are its problem, not yours.
    reactor.QuicHandle = (r, conn) =>
        new Nghttp3Connection(conn).RunStreamingAsync(async request =>
        {
            // Streamed dispatch: you are running while the body is still on the wire. Every read
            // credits the peer's flow-control window, which is what throttles a fast sender
            // instead of buffering it - memory is bound by one window, not the upload size.
            while (true)
            {
                ReadOnlyMemory<byte> chunk = await request.BodyReader!.ReadAsync();
                if (chunk.IsEmpty) break;
                // a real app parses or stores the chunk here
            }

            // request.Path is post-QPACK BYTES. Route with SequenceEqual against a u8 literal
            // rather than decoding it to a string.
            return response;
        });

    threads[id] = new Thread(reactor.Run) { Name = $"reactor-{id}" };
    threads[id].Start();
}

Console.WriteLine($"HTTP/3 on udp :8443 across {config.ReactorCount} reactors (ngtcp2 {QuicEngine.NativeVersion()})");

foreach (Thread thread in threads)
{
    thread.Join();
}

HTTP/3 · pure C#

// A complete HTTP/3 server with NO native h3 code. Paste into a console app's Program.cs.
//   dotnet add package ioxide
//   dotnet add package ioxide.ngtcp2      (QUIC transport)
//   dotnet add package ioxide.http3       (pure-C# HTTP/3)
// Needs cert.pem / key.pem next to the binary.
//   curl --http3-only -k https://127.0.0.1:8443/
using System.Text;
using ioxide;
using ioxide.http3;
using ioxide.ngtcp2;

using var engine = new QuicEngine("cert.pem", "key.pem", cidLength: 8, alpn: ["h3"]);

var config = new ServerConfig
{
    ReactorCount = Environment.ProcessorCount,
    Udp = new UdpOptions { RecvSlots = 16 },
    Quic = new QuicOptions
    {
        Port              = 8443,
        LocalCidLength    = 8,
        ConnectionFactory = engine.CreateFactory(),
    },
};

var threads = new Thread[config.ReactorCount];

for (int id = 0; id < threads.Length; id++)
{
    var reactor = new Reactor(id, config);

    // ioxide.http3 is frame parsing, QPACK (static table + Huffman) and request dispatch, all
    // managed. ngtcp2 still provides the QUIC transport, but nothing above it is native - and it
    // rides any QuicConnection via the stream read surface, so it is engine-agnostic.
    reactor.QuicHandle = (r, conn) => new Http3Connection(conn).RunAsync(async request =>
    {
        // Same streaming surface as the nghttp3 flavour: the body arrives under flow-control
        // pacing, so memory is bound by one window rather than the size of the upload.
        long total = 0;
        while (true)
        {
            ReadOnlyMemory<byte> chunk = await request.BodyReader!.ReadAsync();
            if (chunk.IsEmpty) break;
            total += chunk.Length;
        }

        return Http3Response.Text(
            $"hello {Encoding.ASCII.GetString(request.Path.Span)} - {total} body bytes, pure-C# HTTP/3\n");
    });

    threads[id] = new Thread(reactor.Run) { Name = $"reactor-{id}" };
    threads[id].Start();
}

Console.WriteLine($"pure-C# HTTP/3 on udp :8443 across {config.ReactorCount} reactors");

foreach (Thread thread in threads)
{
    thread.Join();
}

HTTP client · reverse proxy

ioxide + ioxide.httpclient 
// A complete reverse proxy. Paste into a console app's Program.cs and run it.
//   dotnet add package ioxide
//   dotnet add package ioxide.httpclient
// Both hops - the inbound connection and the outbound call - run on this reactor's ring and
// resume inline, so a proxied request never leaves the thread it arrived on.
using System.Text;
using ioxide;
using ioxide.http11;        // HttpClientResponse lives here
using ioxide.httpclient;
using ioxide.utils;

var config = new ServerConfig
{
    ReactorCount = Environment.ProcessorCount,
    Tcp = new TcpOptions { Port = 8080 },
};

var upstream = new RingHttpClientOptions
{
    Host = "127.0.0.1",        // IPv4 literal: a DNS lookup would block the reactor
    Port = 8081,
    PoolSize = 8,              // connections per protocol pool, per reactor

    // Negotiate: start on HTTP/1.1 and switch to HTTP/3 once the origin advertises it via
    // Alt-Svc. Http1Only / Http2Only (h2c, prior knowledge) / Http3Only pin it instead.
    Policy = HttpProtocolPolicy.Negotiate,
};

var threads = new Thread[config.ReactorCount];

for (int id = 0; id < threads.Length; id++)
{
    var reactor = new Reactor(id, config);

    // Opened on the reactor thread, so the outbound sockets ride this reactor's ring too.
    reactor.OnStart = r => RingHttpClient.Start(r, upstream);

    reactor.TcpHandle = async (r, conn) =>
    {
        RingHttpClient http = r.GetService<RingHttpClient>();

        try
        {
            while (true)
            {
                RecvSnapshot snapshot = await conn.ReadAsync();

                while (conn.TryGetItem(snapshot, out SpscRecvRing.Item item))
                {
                    if (item.HasBuffer) conn.ReturnBuffer(in item);
                }

                try
                {
                    // The outbound call. Same ring, same thread - this await resumes inline.
                    // The response owns its bytes, so dispose it when you are done with them.
                    using HttpClientResponse response = await http.GetAsync("/api/thing");

                    conn.Write(Encoding.ASCII.GetBytes(
                        $"HTTP/1.1 {response.Status} OK\r\nContent-Length: {response.Body.Length}\r\n\r\n"));
                    conn.Write(response.Body.Span);   // bytes straight through, no decode
                }
                catch (Exception e)
                {
                    // A dead origin surfaces here rather than hanging: the pool bounds the whole
                    // acquire, so you get an exception and can answer with a 502.
                    byte[] message = Encoding.ASCII.GetBytes($"upstream: {e.Message}");
                    conn.Write(Encoding.ASCII.GetBytes(
                        $"HTTP/1.1 502 Bad Gateway\r\nContent-Length: {message.Length}\r\n\r\n"));
                    conn.Write(message);
                }

                await conn.FlushAsync();

                if (snapshot.IsClosed) return;
                conn.ResetRead();
            }
        }
        finally
        {
            conn.DecRef();
        }
    };

    threads[id] = new Thread(reactor.Run) { Name = $"reactor-{id}" };
    threads[id].Start();
}

Console.WriteLine($"proxy on :{config.Tcp.Port} -> {upstream.Host}:{upstream.Port}");

foreach (Thread thread in threads)
{
    thread.Join();
}

ioxide + ioxide.pg 
// A complete Postgres-backed ioxide server. Paste into a console app's Program.cs and run it.
//   dotnet add package ioxide
//   dotnet add package ioxide.pg
//   docker run --rm -d -p 5432:5432 -e POSTGRES_USER=bench -e POSTGRES_DB=bench \
//     -e POSTGRES_HOST_AUTH_METHOD=trust postgres:18
using System.Text;
using ioxide;
using ioxide.pg;
using ioxide.utils;

var config = new ServerConfig
{
    ReactorCount = Environment.ProcessorCount,
    Tcp = new TcpOptions { Port = 8080 },
};

var pgOptions = new PgOptions
{
    Host = "127.0.0.1",        // IPv4 literal: a DNS lookup would block the reactor
    Port = 5432,
    User = "bench",
    Database = "bench",
    PoolSize = 4,              // per reactor, not global - 12 reactors x 4 = 48 backends
    CommandTimeoutMs = 30_000,
};

var threads = new Thread[config.ReactorCount];

for (int id = 0; id < threads.Length; id++)
{
    var reactor = new Reactor(id, config);

    // OnStart runs ON the reactor thread, so every connection the pool opens belongs to THIS
    // reactor's ring. Start registers the pool as a reactor service; the handler fetches it back
    // with GetService. One pool per reactor, no sharing, no lock.
    reactor.OnStart = r => PgPool.Start(r, pgOptions);

    reactor.TcpHandle = async (r, conn) =>
    {
        PgPool pg = r.GetService<PgPool>();

        try
        {
            while (true)
            {
                RecvSnapshot snapshot = await conn.ReadAsync();

                while (conn.TryGetItem(snapshot, out SpscRecvRing.Item item))
                {
                    if (item.HasBuffer) conn.ReturnBuffer(in item);
                }

                try
                {
                    // io_uring send + recv to Postgres, on the same ring that accepted the
                    // request. The continuation resumes inline, on this thread.
                    PgResult result = await pg.QueryAsync("SELECT 42");

                    string body = $"db={result.Value}";
                    conn.Write(Encoding.ASCII.GetBytes(
                        $"HTTP/1.1 200 OK\r\nContent-Type: text/plain\r\nContent-Length: {body.Length}\r\n\r\n{body}"));
                }
                catch (PgException e)
                {
                    // A server error is just an exception. The connection goes back to the pool
                    // usable - Postgres rejected a query, it did not break the socket.
                    Console.Error.WriteLine($"pg: {e.Message}");
                    conn.Write("HTTP/1.1 500 Internal Server Error\r\nContent-Length: 0\r\n\r\n"u8);
                }

                await conn.FlushAsync();

                if (snapshot.IsClosed) return;
                conn.ResetRead();
            }
        }
        finally
        {
            conn.DecRef();
        }
    };

    threads[id] = new Thread(reactor.Run) { Name = $"reactor-{id}" };
    threads[id].Start();
}

Console.WriteLine($"listening on :{config.Tcp.Port} -> pg {pgOptions.Host}:{pgOptions.Port}");

foreach (Thread thread in threads)
{
    thread.Join();
}

    PoolSize = 8,              // per reactor
    CommandTimeoutMs = 30_000,
};

for (int id = 0; id < threads.Length; id++)
{
    var reactor = new Reactor(id, config);

    // Opened on the reactor thread, so the pool's sockets ride this reactor's ring.
    reactor.OnStart = r => RedisPool.Start(r, redisOptions);

    reactor.TcpHandle = async (r, conn) =>
    {
        RedisPool redis = r.GetService<RedisPool>();

        try
        {
            while (true)
            {
                RecvSnapshot snapshot = await conn.ReadAsync();

                while (conn.TryGetItem(snapshot, out SpscRecvRing.Item item))
                {
                    if (item.HasBuffer) conn.ReturnBuffer(in item);
                }

                // Cache-aside, both halves on this reactor's ring. Typed helper for the hot path.
                string? cached = await redis.GetAsync("user:1");

                if (cached is null)
                {
                    // Generic command surface: anything RESP2 speaks, any argument shape.
                    await redis.ExecuteAsync("SET", "user:1", "ada");
                    cached = "ada";
                }

                RespValue hits = await redis.ExecuteAsync("INCR", "hits");

                string body = $"{cached} ({hits.Integer} hits)";
                conn.Write(Encoding.ASCII.GetBytes(
                    $"HTTP/1.1 200 OK\r\nContent-Type: text/plain\r\nContent-Length: {body.Length}\r\n\r\n{body}"));
                await conn.FlushAsync();

                if (snapshot.IsClosed) return;
                conn.ResetRead();
            }
        }
        finally
        {
            conn.DecRef();
        }
    };

    threads[id] = new Thread(reactor.Run) { Name = $"reactor-{id}" };
    threads[id].Start();
}

// A complete static file server. Paste into a console app's Program.cs and run it.
//   dotnet add package ioxide
//   dotnet add package ioxide.file
using System.Text;
using ioxide;
using ioxide.file;
using ioxide.utils;

// The asset cache opens every file ONCE and shares the descriptors across all reactors: they are
// stable and read positionally, so nothing is locked. Built before the reactors start.
var assets = new StaticAssets("/srv/www", maxCachedFileBytes: 256 * 1024);

var config = new ServerConfig
{
    ReactorCount = Environment.ProcessorCount,
    Tcp = new TcpOptions { Port = 8080 },
};

var threads = new Thread[config.ReactorCount];

for (int id = 0; id < threads.Length; id++)
{
    var reactor = new Reactor(id, config);

    reactor.OnStart = r =>
    {
        r.AddService(assets);                                          // the shared snapshot
        AssetReader.CreatePool(r, readers: 4, bufferBytes: 1 << 20);   // per-reactor ring readers
    };

    reactor.TcpHandle = async (r, conn) =>
    {
        StaticAssets snapshot = r.GetService<StaticAssets>();
        RingPool<AssetReader> readers = r.GetService<RingPool<AssetReader>>();

        try
        {
            while (true)
            {
                RecvSnapshot recv = await conn.ReadAsync();

                // The lease pins the snapshot for the whole request, so a concurrent Reload()
                // cannot free the fd or the baked response out from under an in-flight send.
                using (StaticAssets.Lease lease = snapshot.Acquire())
                {
                    // Resolve the target while the recv bytes are still valid. The lookup is
                    // span-based, so a cache hit allocates no string at all.
                    bool found = false;
                    AssetCache.Asset asset = default;

                    while (conn.TryGetItem(recv, out SpscRecvRing.Item item))
                    {
                        if (item.HasBuffer)
                        {
                            if (!found && TryReadTarget(item.AsSpan(), out ReadOnlySpan<byte> target))
                            {
                                found = lease.TryGet(target, out asset);
                            }
                            conn.ReturnBuffer(in item);
                        }
                    }

                    // Revalidated against disk (size + mtime + inode): the baked response is the
                    // hot path only while the file is unchanged. An edit or an atomic rename is
                    // served live from the new inode instead of stale from RAM.
                    bool fresh = false, exists = false;
                    if (found)
                    {
                        fresh = AssetCache.IsFresh(asset, out exists, out _);
                    }

                    if (found && exists && fresh && asset.Response != 0)
                    {
                        // No open, no read, no formatting - a complete HTTP response already in
                        // native memory. Just send it.
                        await SendNativeAsync(conn, asset.Response, asset.ResponseLength);
                    }
                    else if (found && exists)
                    {
                        // Too big to pin: read it off the ring at advancing offsets.
                        AssetReader reader = await readers.RentAsync();
                        try
                        {
                            Span<byte> header = stackalloc byte[256];
                            conn.Write(header[..AssetCache.WriteResponseHeader(header, asset.Path, (int)asset.Length)]);

                            long offset = 0;
                            while (offset < asset.Length)
                            {
                                int read = await reader.ReadAsync(asset.Fd, offset);   // io_uring positional read
                                if (read <= 0) break;
                                await SendNativeAsync(conn, reader.Buffer, read);
                                offset += read;
                            }
                        }
                        finally
                        {
                            readers.Return(reader);
                        }
                    }
                    else
                    {
                        conn.Write("HTTP/1.1 404 Not Found\r\nContent-Length: 0\r\n\r\n"u8);
                        await conn.FlushAsync();
                    }
                }

                if (recv.IsClosed) return;
                conn.ResetRead();
            }
        }
        finally
        {
            conn.DecRef();
        }
    };

    threads[id] = new Thread(reactor.Run) { Name = $"reactor-{id}" };
    threads[id].Start();
}

Console.WriteLine($"serving {assets.Count} files from {assets.RootDir} on :{config.Tcp.Port}");

// Push native memory through the write slab in chunks.
static async Task SendNativeAsync(TcpConnection conn, nint data, int length)
{
    const int Chunk = 12 * 1024;
    int sent = 0;

    while (true)
    {
        int take = Math.Min(length - sent, Chunk);
        unsafe
        {
            conn.Write(new ReadOnlySpan<byte>((void*)(data + sent), take));
        }
        await conn.FlushAsync();
        sent += take;

        if (sent >= length) return;
    }
}

// "GET /css/app.css?v=1 HTTP/1.1" -> "/css/app.css"
static bool TryReadTarget(ReadOnlySpan<byte> request, out ReadOnlySpan<byte> target)
{
    target = default;

    int firstSpace = request.IndexOf((byte)' ');
    if (firstSpace < 0) return false;

    ReadOnlySpan<byte> afterMethod = request[(firstSpace + 1)..];
    int secondSpace = afterMethod.IndexOf((byte)' ');
    if (secondSpace < 0) return false;

    target = afterMethod[..secondSpace];

    int query = target.IndexOf((byte)'?');
    if (query >= 0) target = target[..query];

    return true;
}

// A complete TLS server with kernel offload. Paste into a console app's Program.cs and run it.
//   dotnet add package ioxide
//   dotnet add package ioxide.tls
// Needs the Linux 'tls' module (sudo modprobe tls), OpenSSL 3, and cert.pem / key.pem.
//   curl -k https://127.0.0.1:8443/
using ioxide;
using ioxide.tls;
using ioxide.utils;

var config = new ServerConfig
{
    ReactorCount = Environment.ProcessorCount,
    Tcp = new TcpOptions { Port = 8443 },
};

ReadOnlyMemory<byte> ok = "HTTP/1.1 200 OK\r\nContent-Type: text/plain\r\nContent-Length: 2\r\n\r\nok"u8.ToArray();

var threads = new Thread[config.ReactorCount];

for (int id = 0; id < threads.Length; id++)
{
    var reactor = new Reactor(id, config);

    reactor.OnStart = r => TlsService.Start(r, new TlsOptions
    {
        CertificatePath = "cert.pem",
        KeyPath         = "key.pem",
        Alpn            = "http/1.1",
    });

    reactor.TcpHandle = async (r, conn) =>
    {
        TlsSession? tls = null;

        try
        {
            // The OpenSSL handshake is driven over the ring, so it awaits like everything else.
            // Once it completes, kernel TLS takes over transmit: everything you write below is
            // PLAINTEXT and the kernel produces the records.
            tls = await r.GetService<TlsService>().AcceptAsync(conn);

            // A request can ride in with the handshake's final flight - answer it first.
            if (tls.DrainPlaintext().Length > 0)
            {
                conn.Write(ok.Span);
                await conn.FlushAsync();
            }

            while (true)
            {
                RecvSnapshot snapshot = await conn.ReadAsync();

                int got = 0;
                while (conn.TryGetItem(snapshot, out SpscRecvRing.Item item))
                {
                    if (item.HasBuffer)
                    {
                        // Inbound still needs decrypting; only transmit is offloaded.
                        got += Decrypt(tls, in item);
                        conn.ReturnBuffer(in item);
                    }
                }

                if (got > 0)
                {
                    conn.Write(ok.Span);   // plaintext in; the kernel encrypts on send
                    await conn.FlushAsync();
                }

                if (snapshot.IsClosed || tls.Closed) return;
                conn.ResetRead();
            }
        }
        catch (Exception e)
        {
            // Handlers are invoked fire-and-forget, so an unobserved handshake error would vanish
            // silently - a missing 'tls' kernel module shows up exactly here.
            Console.Error.WriteLine($"tls: {e.Message}");
        }
        finally
        {
            tls?.Dispose();
            conn.DecRef();
        }
    };

    threads[id] = new Thread(reactor.Run) { Name = $"reactor-{id}" };
    threads[id].Start();
}

Console.WriteLine($"TLS on :{config.Tcp.Port} across {config.ReactorCount} reactors (kTLS transmit offload)");

foreach (Thread thread in threads)
{
    thread.Join();
}

static unsafe int Decrypt(TlsSession tls, in SpscRecvRing.Item item)
{
    fixed (byte* cipher = item.AsSpan())
    {
        return tls.Decrypt(cipher, item.AsSpan().Length).Length;
    }
}
